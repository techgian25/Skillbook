Day 13: Local Storage

What is Local Storage?

Local Storage is a web storage API that allows you to store key-value pairs in the browser. Data persists even after the browser is closed and reopened.

Local Storage vs Session Storage:

  • Local Storage: Persists until explicitly cleared
  • Session Storage: Cleared when tab/window is closed
  • Both store strings only (need to serialize objects)
  • Both are synchronous (block the main thread)
  • Storage limit: ~5-10MB per origin
Basic Operations

The localStorage API provides simple methods to store, retrieve, and remove data.

Storing Data:

// Store simple values
localStorage.setItem('username', 'john_doe');
localStorage.setItem('theme', 'dark');
localStorage.setItem('fontSize', '16');

// Direct property access (alternative)
localStorage.username = 'john_doe';
localStorage['theme'] = 'dark';

Retrieving Data:

// Get stored values
const username = localStorage.getItem('username');
const theme = localStorage.getItem('theme');

// Direct property access
const fontSize = localStorage.fontSize;
const language = localStorage['language'];

// Handle non-existent keys
const userId = localStorage.getItem('userId') || 'anonymous';
const settings = localStorage.getItem('settings') || '{}';

// Check if key exists
if (localStorage.getItem('token')) {
    // User is logged in
}

Removing Data:

// Remove specific item
localStorage.removeItem('tempData');

// Clear all data
localStorage.clear();

// Remove multiple items
const keysToRemove = ['cache1', 'cache2', 'temp'];
keysToRemove.forEach(key => localStorage.removeItem(key));
Storing Complex Data

Local Storage only stores strings, so you need to serialize complex data types like objects and arrays.

JSON Serialization:

// Store objects
const user = {
    id: 123,
    name: 'John Doe',
    email: 'john@example.com',
    preferences: {
        theme: 'dark',
        notifications: true
    }
};

localStorage.setItem('user', JSON.stringify(user));

// Store arrays
const todos = [
    { id: 1, text: 'Learn JavaScript', completed: true },
    { id: 2, text: 'Build a project', completed: false },
    { id: 3, text: 'Deploy to production', completed: false }
];

localStorage.setItem('todos', JSON.stringify(todos));

// Retrieve and parse
const storedUser = JSON.parse(localStorage.getItem('user'));
const storedTodos = JSON.parse(localStorage.getItem('todos'));

// Handle parsing errors
function getStoredData(key, defaultValue = null) {
    try {
        const item = localStorage.getItem(key);
        return item ? JSON.parse(item) : defaultValue;
    } catch (error) {
        console.log(`Error parsing ${key} from localStorage:`, error);
        return defaultValue;
    }
}

const userData = getStoredData('user', {});
const todoList = getStoredData('todos', []);
Storage Events

The storage event fires when data is modified in localStorage (but not in the same window/tab that made the change). 

// Listen for storage changes from other tabs/windows
window.addEventListener('storage', function(event) {
    console.log('Storage changed:', {
        key: event.key,
        oldValue: event.oldValue,
        newValue: event.newValue,
        url: event.url,
        storageArea: event.storageArea
    });
    
    // React to specific changes
    if (event.key === 'theme') {
        applyTheme(event.newValue);
    }
    
    if (event.key === 'user') {
        updateUserInterface(event.newValue);
    }
});

// Example: Theme synchronization across tabs
function applyTheme(theme) {
    document.body.className = theme;
}

// Initial theme application
const currentTheme = localStorage.getItem('theme') || 'light';
applyTheme(currentTheme);
Practical Examples

Common use cases for localStorage in web applications.

User Preferences:

class UserPreferences {
    static get(key, defaultValue) {
        try {
            const item = localStorage.getItem(key);
            return item ? JSON.parse(item) : defaultValue;
        } catch {
            return defaultValue;
        }
    }
    
    static set(key, value) {
        try {
            localStorage.setItem(key, JSON.stringify(value));
        } catch (error) {
            console.log('Failed to save preference:', error);
        }
    }
    
    static getTheme() {
        return this.get('theme', 'light');
    }
    
    static setTheme(theme) {
        this.set('theme', theme);
        this.applyTheme(theme);
    }
    
    static applyTheme(theme) {
        document.body.className = theme;
    }
}

// Usage
document.addEventListener('DOMContentLoaded', () => {
    const theme = UserPreferences.getTheme();
    UserPreferences.applyTheme(theme);
});

Form Data Persistence:

class FormPersistence {
    constructor(formId) {
        this.formId = formId;
        this.form = document.getElementById(formId);
        this.loadFormData();
        this.setupAutoSave();
    }
    
    loadFormData() {
        const savedData = localStorage.getItem(`form_${this.formId}`);
        if (savedData) {
            const data = JSON.parse(savedData);
            Object.keys(data).forEach(key => {
                const input = this.form.elements[key];
                if (input) {
                    input.value = data[key];
                }
            });
        }
    }
    
    saveFormData() {
        const data = {};
        Array.from(this.form.elements).forEach(element => {
            if (element.name && element.value) {
                data[element.name] = element.value;
            }
        });
        localStorage.setItem(`form_${this.formId}`, JSON.stringify(data));
    }
    
    setupAutoSave() {
        this.form.addEventListener('input', () => {
            this.saveFormData();
        });
    }
    
    clearSavedData() {
        localStorage.removeItem(`form_${this.formId}`);
    }
}

// Usage
const contactForm = new FormPersistence('contact-form');
Shopping Cart Example

A complete example of using localStorage for a shopping cart. 

class ShoppingCart {
    constructor() {
        this.items = this.loadCart();
    }
    
    loadCart() {
        try {
            const cart = localStorage.getItem('shoppingCart');
            return cart ? JSON.parse(cart) : [];
        } catch (error) {
            console.log('Error loading cart:', error);
            return [];
        }
    }
    
    saveCart() {
        try {
            localStorage.setItem('shoppingCart', JSON.stringify(this.items));
        } catch (error) {
            console.log('Error saving cart:', error);
        }
    }
    
    addItem(product) {
        const existingItem = this.items.find(item => item.id === product.id);
        
        if (existingItem) {
            existingItem.quantity += 1;
        } else {
            this.items.push({
                id: product.id,
                name: product.name,
                price: product.price,
                quantity: 1
            });
        }
        
        this.saveCart();
        this.updateUI();
    }
    
    removeItem(productId) {
        this.items = this.items.filter(item => item.id !== productId);
        this.saveCart();
        this.updateUI();
    }
    
    getTotal() {
        return this.items.reduce((total, item) => {
            return total + (item.price * item.quantity);
        }, 0);
    }
    
    getItemCount() {
        return this.items.reduce((count, item) => count + item.quantity, 0);
    }
    
    clearCart() {
        this.items = [];
        localStorage.removeItem('shoppingCart');
        this.updateUI();
    }
    
    updateUI() {
        // Update cart count in header
        const cartCount = document.getElementById('cart-count');
        if (cartCount) {
            cartCount.textContent = this.getItemCount();
        }
        
        // Update cart total
        const cartTotal = document.getElementById('cart-total');
        if (cartTotal) {
            cartTotal.textContent = `$${this.getTotal().toFixed(2)}`;
        }
    }
}

// Usage
const cart = new ShoppingCart();

// Add event listeners to "Add to Cart" buttons
document.querySelectorAll('.add-to-cart').forEach(button => {
    button.addEventListener('click', (e) => {
        const product = {
            id: e.target.dataset.productId,
            name: e.target.dataset.productName,
            price: parseFloat(e.target.dataset.productPrice)
        };
        cart.addItem(product);
    });
});
Error Handling and Best Practices

Handling storage limitations and implementing robust localStorage usage.

Storage Quota Handling:

function safeSetItem(key, value) {
    try {
        localStorage.setItem(key, value);
        return true;
    } catch (error) {
        if (error.name === 'QuotaExceededError') {
            console.log('Storage quota exceeded');
            // Try to free up space by removing old data
            clearOldCache();
            try {
                localStorage.setItem(key, value);
                return true;
            } catch {
                return false;
            }
        }
        console.log('Storage error:', error);
        return false;
    }
}

function clearOldCache() {
    // Remove items older than 7 days
    const now = Date.now();
    const keysToRemove = [];
    
    for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        if (key.startsWith('cache_')) {
            try {
                const data = JSON.parse(localStorage.getItem(key));
                if (data.timestamp && (now - data.timestamp) > 7 * 24 * 60 * 60 * 1000) {
                    keysToRemove.push(key);
                }
            } catch {
                // Invalid data, remove it
                keysToRemove.push(key);
            }
        }
    }
    
    keysToRemove.forEach(key => localStorage.removeItem(key));
}

Best Practices:

  • Always handle JSON.parse() errors
  • Check for storage availability before use
  • Use meaningful key names with prefixes
  • Implement data versioning for stored objects
  • Don't store sensitive information
  • Provide fallbacks when localStorage is unavailable
  • Consider using IndexedDB for large/complex data
  • Test with storage disabled in browser dev tools
Session Storage

Session Storage works similarly to localStorage but data is cleared when the tab/window is closed. 

// Session Storage API (same as localStorage)
sessionStorage.setItem('tempData', 'value');
const tempData = sessionStorage.getItem('tempData');
sessionStorage.removeItem('tempData');
sessionStorage.clear();

// Use cases for sessionStorage:
// - Temporary form data
// - Session-specific settings
// - Shopping cart for guest users
// - Wizard/stepper state
// - Authentication tokens (if you want them cleared on tab close)

class SessionManager {
    static set(key, value) {
        try {
            sessionStorage.setItem(key, JSON.stringify(value));
        } catch (error) {
            console.log('Session storage error:', error);
        }
    }
    
    static get(key, defaultValue = null) {
        try {
            const item = sessionStorage.getItem(key);
            return item ? JSON.parse(item) : defaultValue;
        } catch {
            return defaultValue;
        }
    }
    
    static remove(key) {
        sessionStorage.removeItem(key);
    }
    
    static clear() {
        sessionStorage.clear();
    }
}
