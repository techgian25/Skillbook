Day 10: Error Handling

What are Errors?

Errors are problems that occur during the execution of a program. Proper error handling prevents crashes and provides better user experience.

Types of Errors:

  • SyntaxError - Invalid JavaScript syntax
  • ReferenceError - Reference to non-existent variable
  • TypeError - Operation on wrong type
  • RangeError - Number outside valid range
  • URIError - Invalid URI operations
  • EvalError - Error in eval() function
Try-Catch Blocks

The try-catch statement marks a block of statements to try, and specifies a response should an exception be thrown. 

try {
    // Code that might throw an error
    const result = riskyOperation();
    console.log('Success:', result);
} catch (error) {
    // Code to handle the error
    console.log('Error occurred:', error.message);
    // Handle the error appropriately
} finally {
    // Code that always executes
    console.log('Cleanup code here');
}

Error Object Properties:

try {
    throw new Error('Something went wrong');
} catch (error) {
    console.log('Name:', error.name);        // 'Error'
    console.log('Message:', error.message);  // 'Something went wrong'
    console.log('Stack:', error.stack);      // Stack trace
}
Throwing Errors

You can throw your own errors using the throw statement. 

function divide(a, b) {
    if (b === 0) {
        throw new Error('Division by zero is not allowed');
    }
    return a / b;
}

try {
    const result = divide(10, 0);
} catch (error) {
    console.log('Caught error:', error.message);
}

// Throwing different types of errors
function validateAge(age) {
    if (typeof age !== 'number') {
        throw new TypeError('Age must be a number');
    }
    if (age < 0) {
        throw new RangeError('Age cannot be negative');
    }
    if (age > 150) {
        throw new RangeError('Age seems unrealistic');
    }
    return true;
}
Custom Error Classes

You can create custom error classes that extend the built-in Error class. 

class ValidationError extends Error {
    constructor(message, field) {
        super(message);
        this.name = 'ValidationError';
        this.field = field;
    }
}

class NetworkError extends Error {
    constructor(message, statusCode) {
        super(message);
        this.name = 'NetworkError';
        this.statusCode = statusCode;
    }
}

// Usage
function validateUser(user) {
    if (!user.name) {
        throw new ValidationError('Name is required', 'name');
    }
    if (!user.email) {
        throw new ValidationError('Email is required', 'email');
    }
}

try {
    validateUser({ name: 'John' });
} catch (error) {
    if (error instanceof ValidationError) {
        console.log(`Validation failed for ${error.field}: ${error.message}`);
    } else {
        console.log('Unexpected error:', error.message);
    }
}
Error Handling in Promises

Promises have built-in error handling with .catch() and try-catch in async functions.

Promise Error Handling:

const fetchData = () => {
    return new Promise((resolve, reject) => {
        // Simulate API call
        setTimeout(() => {
            const success = Math.random() > 0.5;
            if (success) {
                resolve({ data: 'Success!' });
            } else {
                reject(new Error('API call failed'));
            }
        }, 1000);
    });
};

// Using .catch()
fetchData()
    .then(result => {
        console.log('Data:', result.data);
    })
    .catch(error => {
        console.log('Error:', error.message);
        // Handle error (retry, show message, etc.)
    });

// Chaining errors
fetchData()
    .then(processData)
    .then(saveData)
    .catch(error => {
        console.log('Something failed in the chain:', error.message);
    });

Async/Await Error Handling:

async function fetchUserData() {
    try {
        const response = await fetch('/api/user');
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        const userData = await response.json();
        return userData;
    } catch (error) {
        console.log('Failed to fetch user data:', error.message);
        // Could throw a custom error or return a default value
        throw new Error('Unable to load user data');
    }
}

// Using the function
async function displayUser() {
    try {
        const user = await fetchUserData();
        console.log('User:', user);
    } catch (error) {
        console.log('Could not display user:', error.message);
        // Show error message to user
    }
}
Global Error Handling

You can set up global error handlers to catch unhandled errors and promise rejections.

Window Error Events:

// Global error handler for runtime errors
window.addEventListener('error', function(event) {
    console.log('Global error caught:', event.error);
    // Send error to logging service
    logErrorToService(event.error);
    
    // Prevent default error handling
    event.preventDefault();
});

// Global handler for unhandled promise rejections
window.addEventListener('unhandledrejection', function(event) {
    console.log('Unhandled promise rejection:', event.reason);
    // Handle the rejection
    event.preventDefault();
});

Error Logging:

function logError(error, context = {}) {
    const errorData = {
        message: error.message,
        stack: error.stack,
        timestamp: new Date().toISOString(),
        userAgent: navigator.userAgent,
        url: window.location.href,
        ...context
    };
    
    // Send to error logging service
    console.log('Logging error:', errorData);
    
    // In a real app, you might send this to:
    // - Sentry, LogRocket, Bugsnag, etc.
    // - Your own logging API
}

// Usage
try {
    riskyOperation();
} catch (error) {
    logError(error, { userId: '123', action: 'saveData' });
}
Best Practices

Guidelines for effective error handling in JavaScript applications.

General Best Practices:

  • Always use try-catch for synchronous operations that might fail
  • Handle promise rejections with .catch() or try-catch in async functions
  • Provide meaningful error messages
  • Don't expose sensitive information in error messages
  • Use custom error classes for different types of errors
  • Log errors for debugging but don't show stack traces to users
  • Fail fast - don't continue execution after critical errors

Error Recovery Patterns:

// Retry pattern
async function fetchWithRetry(url, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
        try {
            const response = await fetch(url);
            return await response.json();
        } catch (error) {
            if (i === maxRetries - 1) throw error;
            console.log(`Attempt ${i + 1} failed, retrying...`);
            await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
        }
    }
}

// Fallback pattern
function getUserData() {
    try {
        return JSON.parse(localStorage.getItem('userData'));
    } catch (error) {
        console.log('Could not parse user data, using defaults');
        return { name: 'Guest', preferences: {} };
    }
}

// Graceful degradation
function loadImage(src) {
    return new Promise((resolve, reject) => {
        const img = new Image();
        img.onload = () => resolve(img);
        img.onerror = () => {
            console.log('Image failed to load, using placeholder');
            resolve(createPlaceholderImage());
        };
        img.src = src;
    });
}
