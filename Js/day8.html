Day 8: Asynchronous JavaScript

Synchronous vs Asynchronous

JavaScript is single-threaded, meaning it can only execute one task at a time. Asynchronous programming allows non-blocking operations.

Synchronous (Blocking):

console.log('Start');
console.log('Middle'); // Waits for previous line
console.log('End');
// Output: Start, Middle, End

Asynchronous (Non-blocking):

console.log('Start');
setTimeout(() => console.log('Async operation'), 1000);
console.log('End');
// Output: Start, End, Async operation (after 1 second)
Callbacks

Callbacks are functions passed as arguments to other functions, executed when an asynchronous operation completes. 

// Basic callback
function doSomething(callback) {
    setTimeout(() => {
        console.log('Operation completed');
        callback();
    }, 1000);
}

doSomething(() => {
    console.log('Callback executed');
});

// Callback with data
function fetchData(callback) {
    setTimeout(() => {
        const data = { name: 'John', age: 30 };
        callback(data);
    }, 1000);
}

fetchData((data) => {
    console.log('Received data:', data);
});

Callback Hell:

// Nested callbacks become hard to read
firstAsyncOperation((result1) => {
    secondAsyncOperation(result1, (result2) => {
        thirdAsyncOperation(result2, (result3) => {
            console.log('Final result:', result3);
        });
    });
});
Promises

Promises represent the eventual completion (or failure) of an asynchronous operation and its resulting value.

Creating Promises:

const myPromise = new Promise((resolve, reject) => {
    // Asynchronous operation
    setTimeout(() => {
        const success = true;
        if (success) {
            resolve('Operation successful!');
        } else {
            reject('Operation failed!');
        }
    }, 1000);
});

Using Promises:

myPromise
    .then((result) => {
        console.log('Success:', result);
        return 'Next step';
    })
    .then((nextResult) => {
        console.log('Next:', nextResult);
    })
    .catch((error) => {
        console.log('Error:', error);
    })
    .finally(() => {
        console.log('Always executes');
    });
Async/Await

Async/await is syntactic sugar built on top of promises, making asynchronous code look synchronous.

Async Functions:

async function myAsyncFunction() {
    try {
        const result1 = await someAsyncOperation();
        const result2 = await anotherAsyncOperation(result1);
        console.log('Final result:', result2);
    } catch (error) {
        console.log('Error:', error);
    }
}

// Arrow function version
const myAsyncFunction = async () => {
    // async code here
};

Await Operator:

// Await pauses execution until promise resolves
async function fetchUserData() {
    console.log('Fetching user data...');
    
    const response = await fetch('/api/user');
    const userData = await response.json();
    
    console.log('User data:', userData);
    return userData;
}
Promise Methods

Static methods for working with multiple promises.

Promise.all:

// Waits for all promises to resolve
const promises = [
    fetch('/api/users'),
    fetch('/api/posts'),
    fetch('/api/comments')
];

Promise.all(promises)
    .then((results) => {
        console.log('All results:', results);
    })
    .catch((error) => {
        console.log('One promise failed:', error);
    });

Promise.race:

// Resolves/rejects as soon as first promise settles
const promise1 = new Promise(resolve => setTimeout(() => resolve('First'), 1000));
const promise2 = new Promise(resolve => setTimeout(() => resolve('Second'), 2000));

Promise.race([promise1, promise2])
    .then(result => console.log(result)); // "First"

Promise.allSettled:

// Waits for all promises to settle (resolve or reject)
Promise.allSettled([promise1, promise2])
    .then(results => {
        results.forEach(result => {
            if (result.status === 'fulfilled') {
                console.log('Fulfilled:', result.value);
            } else {
                console.log('Rejected:', result.reason);
            }
        });
    });
Error Handling in Async Code

Proper error handling is crucial in asynchronous operations.

With Promises:

fetch('/api/data')
    .then(response => {
        if (!response.ok) {
            throw new Error('Network response was not ok');
        }
        return response.json();
    })
    .then(data => {
        console.log('Data:', data);
    })
    .catch(error => {
        console.log('Error:', error.message);
    });

With Async/Await:

async function fetchData() {
    try {
        const response = await fetch('/api/data');
        if (!response.ok) {
            throw new Error('Network response was not ok');
        }
        const data = await response.json();
        console.log('Data:', data);
    } catch (error) {
        console.log('Error:', error.message);
        // Handle error appropriately
    }
}
Converting Callbacks to Promises

You can wrap callback-based functions in promises. 

// Callback-based function
function oldApiFunction(callback) {
    setTimeout(() => {
        callback(null, 'Success!');
    }, 1000);
}

// Wrap in promise
function promisifiedApi() {
    return new Promise((resolve, reject) => {
        oldApiFunction((error, result) => {
            if (error) {
                reject(error);
            } else {
                resolve(result);
            }
        });
    });
}

// Use with async/await
async function useApi() {
    try {
        const result = await promisifiedApi();
        console.log(result);
    } catch (error) {
        console.log('Error:', error);
    }
}
